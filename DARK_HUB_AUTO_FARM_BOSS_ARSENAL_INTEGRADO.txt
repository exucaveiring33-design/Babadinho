local function ArsenalCharacter()
    return ArsenalPlayer.Character

local function ArsenalRoot()
    local c = ArsenalCharacter()
    return c and c:FindFirstChild("HumanoidRootPart")
end

local function ArsenalFolder()
    return workspace:FindFirstChild(ARSENAL_PICKUPS)
end

local function ArsenalFindPickup(name)
    local folder = ArsenalFolder()
    if not folder then return nil end

    local exact = folder:FindFirstChild(name)
    if exact and ArsenalWeaponSet[string.lower(exact.Name)] then
        return exact
    end

    local wanted = string.lower(name)
    for _, obj in ipairs(folder:GetChildren()) do
        if string.lower(obj.Name) == wanted then
            return obj
        end
    end
end

local function ArsenalPrimary(obj)
    if not obj then return nil end
    if obj:IsA("BasePart") then return obj end

    if obj:IsA("Model") then
        if obj.PrimaryPart then return obj.PrimaryPart end

        local handle = obj:FindFirstChild("Handle", true)
        if handle and handle:IsA("BasePart") then
            return handle
        end

        local hitbox = obj:FindFirstChild("hitbox", true)
        if hitbox and hitbox:IsA("BasePart") then
            return hitbox
        end

        return obj:FindFirstChildWhichIsA("BasePart", true)
    end

    return obj:FindFirstChildWhichIsA("BasePart", true)
end

local function ArsenalPrompt(obj)
    if not obj then return nil end
    return obj:FindFirstChildWhichIsA("ProximityPrompt", true)
end

local function ArsenalClick(obj)
    if not obj then return nil end
    return obj:FindFirstChildWhichIsA("ClickDetector", true)
end

local function ArsenalMoveNear(part)
    local hrp = ArsenalRoot()
    if not hrp or not part then return false end

    return pcall(function()
        local cf = part.CFrame
        hrp.CFrame =
            cf
            - cf.LookVector * 3
            + Vector3.new(0, 2, 0)
    end)
end

-- Há uma limitação importante: InputHoldBegin/InputHoldEnd de
-- ProximityPrompt não são garantidos como uma API de simulação de
-- entrada no cliente em todas as experiências. O código tenta de forma
-- protegida; se não funcionar, o status explica que é necessário usar
-- a interação normal.
local function ArsenalTryPrompt(prompt)
    if not prompt or not prompt:IsA("ProximityPrompt") then
        return false, "Prompt inválido."
    end

    local ok = pcall(function()
        prompt:InputHoldBegin()

        local duration = tonumber(prompt.HoldDuration) or 0
        task.wait(math.max(duration, 0.1) + 0.08)

        prompt:InputHoldEnd()
    end)

    if ok then
        return true, "ProximityPrompt processado."
    end

    return false, "O Prompt existe, mas a ativação automática foi recusada pelo cliente."
end

local function ArsenalPickupOne(name)
    local pickup = ArsenalFindPickup(name)
    if not pickup then
        return false, "Pickup não encontrado: " .. tostring(name)
    end

    local part = ArsenalPrimary(pickup)
    if part then
        ArsenalMoveNear(part)
        task.wait(0.15)
    end

    local prompt = ArsenalPrompt(pickup)
    if prompt then
        return ArsenalTryPrompt(prompt)
    end

    local detector = ArsenalClick(pickup)
    if detector then
        return false, "Este pickup usa ClickDetector; clique nele normalmente."
    end

    return false, "Nenhum ProximityPrompt/ClickDetector encontrado."
end

local function ArsenalRebuild()
    ArsenalClearButtons()

    local folder = ArsenalFolder()
    local found = {}

    if folder then
        for _,obj in ipairs(folder:GetChildren()) do
            if ArsenalWeaponSet[string.lower(obj.Name)] then
                table.insert(found,obj.Name)
            end
        end
    end

    table.sort(found,function(a,b)
        return string.lower(a) < string.lower(b)
    end)

    for _,name in ipairs(found) do
        ArsenalAddButton(name)
    end

    if #found == 0 then
        ArsenalSelectedLabel.Text = "ARMA: nenhum pickup encontrado"
        ArsenalStatus("workspace.pickups sem armas conhecidas.",false)
        return
    end

    if not ArsenalSelected or not ArsenalWeaponSet[string.lower(ArsenalSelected)] then
        ArsenalSelected = found[1]
        ArsenalSelectedLabel.Text = "ARMA: " .. ArsenalSelected
    end

    ArsenalStatus(#found .. " pickups encontrados.",true)
end

task.spawn(function()
    while not ArsenalDestroyed do
        if ArsenalAuto and not ArsenalBusy then
            local folder = ArsenalFolder()

            if folder then
                for _,obj in ipairs(folder:GetChildren()) do
                    if ArsenalAuto and ArsenalWeaponSet[string.lower(obj.Name)] then
                        local part = ArsenalPrimary(obj)
                        local hrp = ArsenalRoot()

                        if part and hrp then
                            local dist = (part.Position-hrp.Position).Magnitude

                            if dist <= 80 then
                                ArsenalBusy = true
                                ArsenalPickupOne(obj.Name)
                                task.wait(0.15)
                                ArsenalBusy = false
                            end
                        end
                    end
                end
            end
        end

        task.wait(0.5)
    end
end)

local ArsenalFolderNow = ArsenalFolder()

    ArsenalFolderNow.ChildAdded:Connect(function()
        task.wait()
        ArsenalRebuild()
    end)

    ArsenalFolderNow.ChildRemoved:Connect(function()
        task.wait()
        ArsenalRebuild()
    end)
end

local function ArsenalDraggable(handle,object)
    local dragging=false
    local dragStart
    local startPosition
    local dragInput

    handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then

            dragging=true
            dragStart=input.Position
            startPosition=object.Position

            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging=false
                end
            end)
        end
    end)

    handle.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch then
            dragInput=input
        end
    end)

    ArsenalUIS.InputChanged:Connect(function(input)
        if dragging and input == dragInput then
            local delta=input.Position-dragStart

            object.Position=UDim2.new(
                startPosition.X.Scale,
                startPosition.X.Offset+delta.X,
                startPosition.Y.Scale,
                startPosition.Y.Offset+delta.Y
            )
        end
    end)
end
